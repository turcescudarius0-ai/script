-- Roblox Script for "Steal a Brainrot"
-- Visual UI built using Rayfield Library

local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
   Name = "Steal a Brainrot Hub",
   LoadingTitle = "Loading Script...",
   LoadingSubtitle = "by Assistant",
   ConfigurationSaving = {
      Enabled = false,
   },
   Discord = {
      Enabled = false
   },
   KeySystem = false
})

-- Services
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- Variables for toggles
local FlySpeed = 50
local Flying = false
local GodModeEnabled = false
local AutoStealEnabled = false

----------------------------------------------------
-- TAB 1: Self
----------------------------------------------------
local SelfTab = Window:CreateTab("Self", 4483362458) -- Title, ImageId

SelfTab:CreateSection("Movement & Stats")

SelfTab:CreateSlider({
   Name = "WalkSpeed",
   Range = {16, 250},
   Increment = 1,
   Suffix = "Speed",
   CurrentValue = 16,
   Flag = "SpeedSlider",
   Callback = function(Value)
      if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
         LocalPlayer.Character.Humanoid.WalkSpeed = Value
      end
   end,
})

SelfTab:CreateSlider({
   Name = "JumpPower",
   Range = {50, 300},
   Increment = 1,
   Suffix = "Power",
   CurrentValue = 50,
   Flag = "JumpSlider",
   Callback = function(Value)
      if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
         LocalPlayer.Character.Humanoid.UseJumpPower = true
         LocalPlayer.Character.Humanoid.JumpPower = Value
      end
   end,
})

SelfTab:CreateToggle({
   Name = "Fly",
   CurrentValue = false,
   Flag = "FlyToggle",
   Callback = function(Value)
      Flying = Value
      local Char = LocalPlayer.Character
      if not Char or not Char:FindFirstChild("HumanoidRootPart") then return end
      local HRP = Char.HumanoidRootPart

      if Flying then
         local BodyVelocity = Instance.new("BodyVelocity")
         BodyVelocity.Name = "FlyVelocity"
         BodyVelocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
         BodyVelocity.Velocity = Vector3.new(0, 0, 0)
         BodyVelocity.Parent = HRP

         local BodyGyro = Instance.new("BodyGyro")
         BodyGyro.Name = "FlyGyro"
         BodyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
         BodyGyro.CFrame = HRP.CFrame
         BodyGyro.Parent = HRP

         task.spawn(function()
            while Flying and task.wait() do
               local Camera = Workspace.CurrentCamera
               local MoveVector = Vector3.new()
               
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.W) then
                  MoveVector = MoveVector + Camera.CFrame.LookVector
               end
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.S) then
                  MoveVector = MoveVector - Camera.CFrame.LookVector
               end
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.A) then
                  MoveVector = MoveVector - Camera.CFrame.RightVector
               end
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.D) then
                  MoveVector = MoveVector + Camera.CFrame.RightVector
               end
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.Space) then
                  MoveVector = MoveVector + Vector3.new(0, 1, 0)
               end
               if game:GetService("UserInputService"):IsKeyDown(Enum.KeyCode.LeftShift) then
                  MoveVector = MoveVector - Vector3.new(0, 1, 0)
               end

               BodyVelocity.Velocity = MoveVector * FlySpeed
               BodyGyro.CFrame = Camera.CFrame
            end
            if BodyVelocity then BodyVelocity:Destroy() end
            if BodyGyro then BodyGyro:Destroy() end
         end)
      else
         if HRP:FindFirstChild("FlyVelocity") then HRP.FlyVelocity:Destroy() end
         if HRP:FindFirstChild("FlyGyro") then HRP.FlyGyro:Destroy() end
      end
   end,
})

SelfTab:CreateSection("Combat & Defense")

SelfTab:CreateToggle({
   Name = "Godmode",
   CurrentValue = false,
   Flag = "GodToggle",
   Callback = function(Value)
      GodModeEnabled = Value
      if GodModeEnabled then
         task.spawn(function()
            while GodModeEnabled and task.wait(0.1) do
               if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                  LocalPlayer.Character.Humanoid.Health = LocalPlayer.Character.Humanoid.MaxHealth
               end
            end
         end)
      end
   end,
})

SelfTab:CreateSection("Automation")

SelfTab:CreateToggle({
   Name = "Auto Steal",
   CurrentValue = false,
   Flag = "AutoStealToggle",
   Callback = function(Value)
      AutoStealEnabled = Value
      if AutoStealEnabled then
         task.spawn(function()
            while AutoStealEnabled and task.wait(0.5) do
               -- Scriptul încearcă să activeze ProximityPrompts sau să atingă obiecte din joc
               for _, v in pairs(Workspace:GetDescendants()) do
                  if v:IsA("ProximityPrompt") then
                     fireproximityprompt(v)
                  end
               end
            end
         end)
      end
   end,
})

SelfTab:CreateButton({
   Name = "Infinite Jump",
   Callback = function()
      game:GetService("UserInputService").JumpRequest:Connect(function()
         if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            LocalPlayer.Character.Humanoid:ChangeState("Jumping")
         end
      end)
   end,
})

----------------------------------------------------
-- TAB 2: Spawn
----------------------------------------------------
local SpawnTab = Window:CreateTab("Spawn", 4483362458)

SpawnTab:CreateSection("Brainrot Pets & Entities")

local function TrySpawn(itemName)
   -- Caută un event de Spawn în ReplicatedStorage specific jocului
   local remote = ReplicatedStorage:FindFirstChild("SpawnItem") 
      or ReplicatedStorage:FindFirstChild("SpawnBrainrot")
      or ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Spawn")
   
   if remote and remote:IsA("RemoteEvent") then
      remote:FireServer(itemName)
      Rayfield:Notify({Title = "Spawn Triggered", Content = "Attempted to spawn: " .. itemName, Duration = 2})
   else
      Rayfield:Notify({Title = "Notice", Content = "Event-ul de spawn nu a fost găsit în ReplicatedStorage.", Duration = 3})
   end
end

SpawnTab:CreateButton({
   Name = "Spawn Strawberry Elephant",
   Callback = function()
      TrySpawn("Strawberry Elephant")
   end,
})

SpawnTab:CreateButton({
   Name = "Spawn Garama and Madumdum",
   Callback = function()
      TrySpawn("Garama and Madumdum")
   end,
})

SpawnTab:CreateButton({
   Name = "Spawn Dragon Cannelloni",
   Callback = function()
      TrySpawn("Dragon Cannelloni")
   end,
})

----------------------------------------------------
-- TAB 3: Settings / Unload
----------------------------------------------------
local SettingsTab = Window:CreateTab("Settings", 4483362458)

SettingsTab:CreateSection("Script Control")

SettingsTab:CreateButton({
   Name = "UNLOAD SCRIPT",
   Callback = function()
      Flying = false
      GodModeEnabled = false
      AutoStealEnabled = false
      
      if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
         if LocalPlayer.Character.HumanoidRootPart:FindFirstChild("FlyVelocity") then
            LocalPlayer.Character.HumanoidRootPart.FlyVelocity:Destroy()
         end
         if LocalPlayer.Character.HumanoidRootPart:FindFirstChild("FlyGyro") then
            LocalPlayer.Character.HumanoidRootPart.FlyGyro:Destroy()
         end
      end
      
      Rayfield:Destroy()
   end,
})
